
CREATE TABLE Users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    bio TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE Posts (
    post_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    content TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES Users(user_id)
);


CREATE TABLE Followers (
    follower_id INT,
    following_id INT,
    followed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (follower_id, following_id),
    FOREIGN KEY (follower_id) REFERENCES Users(user_id),
    FOREIGN KEY (following_id) REFERENCES Users(user_id)
);


CREATE TABLE Likes (
    user_id INT,
    post_id INT,
    liked_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, post_id),
    FOREIGN KEY (user_id) REFERENCES Users(user_id),
    FOREIGN KEY (post_id) REFERENCES Posts(post_id)
);


CREATE TABLE Comments (
    comment_id INT PRIMARY KEY AUTO_INCREMENT,
    post_id INT,
    user_id INT,
    comment_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES Posts(post_id),
    FOREIGN KEY (user_id) REFERENCES Users(user_id)
);

    BEGINNER LEVEL QUERIES
	======================
 1.View all users
SELECT * FROM Users;

2.View all posts
SELECT * FROM Posts;

3.Get posts created by user 1
SELECT * FROM Posts
WHERE user_id = 1;

4.Count total users
SELECT COUNT(*) AS total_users
FROM Users;

5.Count total posts
SELECT COUNT(*) AS total_posts
FROM Posts;

6.Get comments for post 1
SELECT * FROM Comments
WHERE post_id = 1;

7.Get posts sorted by latest
SELECT * FROM Posts
ORDER BY created_at DESC;

    INTERMEDIATE LEVEL QUERIES
	=========================
1.Get username with their posts
SELECT u.username, p.content
FROM Users u
JOIN Posts p ON u.user_id = p.user_id;

2.Count followers of each user
SELECT following_id AS user_id,
       COUNT(follower_id) AS total_followers
FROM Followers
GROUP BY following_id;

3.Count likes per post
SELECT p.post_id, COUNT(l.user_id) AS total_likes
FROM Posts p
LEFT JOIN Likes l ON p.post_id = l.post_id
GROUP BY p.post_id;

4.Get users who liked post 1
SELECT u.username
FROM Users u
JOIN Likes l ON u.user_id = l.user_id
WHERE l.post_id = 1;

5.Get comments with username
SELECT u.username, c.comment_text
FROM Comments c
JOIN Users u ON c.user_id = u.user_id;

6.Find users who have more than 1 post
SELECT user_id, COUNT(*) AS total_posts
FROM Posts
GROUP BY user_id
HAVING COUNT(*) > 1;

7.Get posts with number of comments
SELECT p.post_id, COUNT(c.comment_id) AS total_comments
FROM Posts p
LEFT JOIN Comments c ON p.post_id = c.post_id
GROUP BY p.post_id;

      ADVANCED LEVEL QUERIES
	  ========================
1.Generate User Feed (Posts from followed users)
SELECT p.*
FROM Posts p
JOIN Followers f ON p.user_id = f.following_id
WHERE f.follower_id = 1
ORDER BY p.created_at DESC;

2.Most liked post
SELECT p.post_id, COUNT(l.user_id) AS total_likes
FROM Posts p
LEFT JOIN Likes l ON p.post_id = l.post_id
GROUP BY p.post_id
ORDER BY total_likes DESC
LIMIT 1;

3.Suggest users to follow (not already following)
SELECT u.user_id, u.username
FROM Users u
WHERE u.user_id NOT IN (
    SELECT following_id
    FROM Followers
    WHERE follower_id = 1
)
AND u.user_id != 1;

4.Top 3 most active users (by posts)
SELECT user_id, COUNT(*) AS total_posts
FROM Posts
GROUP BY user_id
ORDER BY total_posts DESC
LIMIT 3;

5.Find users who never posted anything
SELECT u.user_id, u.username
FROM Users u
LEFT JOIN Posts p ON u.user_id = p.user_id
WHERE p.post_id IS NULL;

6.Find mutual followers (User 1 & User 3 follow each other)
SELECT *
FROM Followers f1
JOIN Followers f2
ON f1.follower_id = f2.following_id
AND f1.following_id = f2.follower_id
WHERE f1.follower_id = 1
AND f1.following_id = 3;

7.Rank users by number of followers (Window Function – MySQL 8+)
SELECT following_id,
       COUNT(follower_id) AS total_followers,
       RANK() OVER (ORDER BY COUNT(follower_id) DESC) AS rank_position
FROM Followers
GROUP BY following_id;
